<script>

function generateComic() {

    const prompt = document.getElementById("prompt").value.trim();
    const genre = document.getElementById("genre").value;
    const panelCount = parseInt(document.getElementById("panels").value);

    if (prompt === "") {
        alert("Please enter a story idea.");
        return;
    }

    document.getElementById("loading").style.display = "block";
    document.getElementById("comicOutput").style.display = "none";

    setTimeout(function () {

        const grid = document.getElementById("comicGrid");

        grid.innerHTML = "";

        document.getElementById("comicTitle").innerText =
            "🌟 " + genre + " Adventure";

        const scenes = [
            {
                emoji: "🏫",
                scene: "Aisha enters the school library and notices something unusual.",
                dialogue: "Aisha: Something feels different today..."
            },
            {
                emoji: "📖",
                scene: "She discovers an old mysterious book hidden behind a shelf.",
                dialogue: "Aisha: I have never seen this book before!"
            },
            {
                emoji: "✨",
                scene: "The book suddenly begins to glow with magical light.",
                dialogue: "Aisha: What is happening?"
            },
            {
                emoji: "🤖",
                scene: "A friendly robot appears from inside the magical book.",
                dialogue: "Robo: Hello, Aisha! I need your help."
            },
            {
                emoji: "🌌",
                scene: "Aisha and Robo enter a strange magical world.",
                dialogue: "Aisha: This place is incredible!"
            },
            {
                emoji: "⚡",
                scene: "They face a difficult challenge and work together to solve it.",
                dialogue: "Robo: We can solve this together!"
            },
            {
                emoji: "🌈",
                scene: "Their teamwork saves the magical world.",
                dialogue: "Aisha: We did it!"
            },
            {
                emoji: "🏠",
                scene: "Aisha returns home with a new understanding of courage and friendship.",
                dialogue: "Aisha: Today was the greatest adventure ever."
            }
        ];

        for (let i = 0; i < panelCount; i++) {

            const data = scenes[i];

            const panel = document.createElement("div");

            panel.className = "panel";

            panel.innerHTML = `
                <div class="panel-image">
                    ${data.emoji}
                </div>

                <div class="panel-content">

                    <div class="panel-number">
                        PANEL ${i + 1}
                    </div>

                    <div class="scene">
                        <strong>Scene:</strong><br>
                        ${data.scene}
                    </div>

                    <div class="dialogue">
                        ${data.dialogue}
                    </div>

                </div>
            `;

            grid.appendChild(panel);
        }

        document.getElementById("loading").style.display = "none";
        document.getElementById("comicOutput").style.display = "block";

    }, 1500);
}

</script>
